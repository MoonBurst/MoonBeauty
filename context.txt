pub fn hex_to_color(hex: &str) -> Color {
    let clean = hex.trim().trim_start_matches('#');
    if clean.len() == 6 {
        let r = u8::from_str_radix(&clean[0..2], 16).unwrap_or(0);
        let g = u8::from_str_radix(&clean[2..4], 16).unwrap_or(0);
        let b = u8::from_str_radix(&clean[4..6], 16).unwrap_or(0);
        Color::from_rgb8(r, g, b)
    } else if clean.len() == 8 {
        let r = u8::from_str_radix(&clean[0..2], 16).unwrap_or(0);
        let g = u8::from_str_radix(&clean[2..4], 16).unwrap_or(0);
        let b = u8::from_str_radix(&clean[4..6], 16).unwrap_or(0);
        let a = u8::from_str_radix(&clean[6..8], 16).unwrap_or(255);
        Color::from_rgba8(r, g, b, (a as f32) / 255.0)
    } else {
        Color::WHITE
    }
}

pub fn find_friend_profiles_path() -> PathBuf {
    let home = dirs::home_dir().unwrap_or_else(|| PathBuf::from("/tmp"));
    let candidates = [
        PathBuf::from("resources/friend_profiles.json"),
        home.join("projects/Moon-Beauty-Shell/resources/friend_profiles.json"),
        dirs::config_dir().unwrap_or_else(|| PathBuf::from("/tmp")).join("moon-beauty-shell/resources/friend_profiles.json"),
    ];

    for path in &candidates {
        if path.exists() {
            return path.clone();
        }
    }
    PathBuf::from("resources/friend_profiles.json")
}

pub fn load_friend_profiles() -> Vec<FriendProfile> {
    let path = find_friend_profiles_path();
    if path.exists() {
        if let Ok(content) = fs::read_to_string(&path) {
            if let Ok(list) = serde_json::from_str::<Vec<FriendProfile>>(&content) {
                return list;
            }
        }
    }
    Vec::new()
}

pub fn save_friend_profiles(profiles: &[FriendProfile]) -> Result<(), std::io::Error> {
    let path = find_friend_profiles_path();
    if let Some(parent) = path.parent() {
        fs::create_dir_all(parent)?;
    }
    let json = serde_json::to_string_pretty(profiles)?;
    fs::write(path, json)?;
    Ok(())
}

impl ShellSettings {
    pub fn config_path() -> PathBuf {
        let config_dir = dirs::config_dir().unwrap_or_else(|| PathBuf::from("/tmp"));
        config_dir.join("moon-beauty-shell").join("settings.json")
    }

    pub fn get_window_size(&self, id: &str, def_w: u32, def_h: u32) -> (f32, f32) {
        if let Some(win) = self.standalone_windows.get(id) {
            (win.width as f32, win.height as f32)
        } else {
            (def_w as f32, def_h as f32)
        }
    }

    pub fn load() -> Self {
        let path = Self::config_path();
        if path.exists() {
            if let Ok(content) = fs::read_to_string(&path) {
                if let Ok(mut settings) = serde_json::from_str::<ShellSettings>(&content) {
                    let disk_profiles = load_friend_profiles();
                    if !disk_profiles.is_empty() {
                        settings.friend_profiles = disk_profiles;
                    }
                    return settings;
                }
            }
        }
        let mut def = Self::default();
        def.friend_profiles = load_friend_profiles();
        def
    }

    pub fn save(&self) -> Result<(), std::io::Error> {
        let path = Self::config_path();
        if let Some(parent) = path.parent() {
            fs::create_dir_all(parent)?;
        }
        let json = serde_json::to_string_pretty(self)?;
        let tmp_path = path.with_extension("tmp");
        fs::write(&tmp_path, &json)?;
        fs::rename(tmp_path, &path)?;

        let _ = save_friend_profiles(&self.friend_profiles);
        Ok(())
    }
}

use iced::mouse::Cursor;
use iced::widget::canvas::{self, Canvas, Frame, Geometry, Path, Stroke};
use iced::{Color, Element, Length, Point, Rectangle, Size, Theme};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ShapeType {
    Rounded(u16),
    Flat,
    Slant { left: SlantSide, right: SlantSide },
    Hexagon { cut: u16 },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlantSide {
    Left,
    Right,
}

#[derive(Debug, Clone)]
pub struct ShapeBox {
    pub shape: ShapeType,
    pub background: Color,
    pub border_color: Color,
    pub border_width: f32,
    pub slant_width: f32,
    pub width: Length,
    pub height: Length,
}

impl Default for ShapeBox {
    fn default() -> Self {
        Self {
            shape: ShapeType::Slant {
                left: SlantSide::Left,
                right: SlantSide::Right,
            },
            background: Color::from_rgb8(15, 15, 15),
            border_color: Color::from_rgb8(0, 51, 153),
            border_width: 2.0,
            slant_width: 12.0,
            width: Length::Fill,
            height: Length::Fill,
        }
    }
}

impl ShapeBox {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn shape(mut self, shape: ShapeType) -> Self {
        self.shape = shape;
        self
    }

    pub fn background(mut self, color: Color) -> Self {
        self.background = color;
        self
    }

    pub fn border(mut self, color: Color, width: f32) -> Self {
        self.border_color = color;
        self.border_width = width;
        self
    }

    pub fn slant_width(mut self, width: f32) -> Self {
        self.slant_width = width;
        self
    }

    pub fn size(mut self, width: Length, height: Length) -> Self {
        self.width = width;
        self.height = height;
        self
    }

    pub fn view<Message: 'static>(self) -> Element<'static, Message> {
        let w = self.width;
        let h = self.height;
        Canvas::new(self)
            .width(w)
            .height(h)
            .into()
    }
}

impl<Message> canvas::Program<Message> for ShapeBox {
    type State = ();

    fn draw(
        &self,
        _state: &Self::State,
        renderer: &iced::Renderer,
        _theme: &Theme,
        bounds: Rectangle,
        _cursor: Cursor,
    ) -> Vec<Geometry> {
        let mut frame = Frame::new(renderer, bounds.size());
        let w = bounds.width;
        let h = bounds.height;
        let half = self.border_width / 2.0;

        if w <= half * 2.0 || h <= half * 2.0 {
            return vec![frame.into_geometry()];
        }

        let path = match self.shape {
            ShapeType::Flat => Path::rectangle(
                Point::new(half, half),
                Size::new(w - self.border_width, h - self.border_width),
            ),
            ShapeType::Rounded(r) => {
                let radius = (r as f32).min(w / 2.0).min(h / 2.0);
                Path::rounded_rectangle(
                    Point::new(half, half),
                    Size::new(w - self.border_width, h - self.border_width),
                    radius.into(),
                )
            }
            ShapeType::Slant { left, right } => {
                let sw = self.slant_width.min(w / 2.0);
                let x1 = if left == SlantSide::Right { sw + half } else { half };
                let x2 = if left == SlantSide::Left { sw + half } else { half };
                let x3 = if right == SlantSide::Left { w - sw - half } else { w - half };
                let x4 = if right == SlantSide::Right { w - sw - half } else { w - half };

                Path::new(|builder| {
                    builder.move_to(Point::new(x1, half));
                    builder.line_to(Point::new(x3, half));
                    builder.line_to(Point::new(x4, h - half));
                    builder.line_to(Point::new(x2, h - half));
                    builder.close();
                })
            }
            ShapeType::Hexagon { cut } => {
                let c = (cut as f32).min(w * 0.35).min(h * 0.45);
                Path::new(|builder| {
                    builder.move_to(Point::new(c + half, half));
                    builder.line_to(Point::new(w - c - half, half));
                    builder.line_to(Point::new(w - half, c + half));
                    builder.line_to(Point::new(w - half, h - c - half));
                    builder.line_to(Point::new(w - c - half, h - half));
                    builder.line_to(Point::new(c + half, h - half));
                    builder.line_to(Point::new(half, h - c - half));
                    builder.line_to(Point::new(half, c + half));
                    builder.close();
                })
            }
        };

        if self.background.a > 0.0 {
            frame.fill(&path, self.background);
        }

        if self.border_width > 0.0 && self.border_color.a > 0.0 {
            frame.stroke(
                &path,
                Stroke::default()
                    .with_color(self.border_color)
                    .with_width(self.border_width),
            );
        }

        vec![frame.into_geometry()]
    }
}

pub fn get_socket_path() -> PathBuf {
    let runtime_dir = std::env::var("XDG_RUNTIME_DIR")
        .unwrap_or_else(|_| format!("/tmp/user-{}", std::process::id()));
    PathBuf::from(runtime_dir).join("moon-beauty-shell.sock")
}

pub async fn send_client_command(cmd: IpcCommand) -> Result<String, Box<dyn std::error::Error + Send + Sync>> {
    let path = get_socket_path();
    let mut stream = UnixStream::connect(path).await?;
    let encoded = serde_json::to_vec(&cmd)?;
    stream.write_all(&encoded).await?;
    stream.shutdown().await?;

    let mut response = String::new();
    stream.read_to_string(&mut response).await?;
    Ok(response)
}

pub struct IpcServer {
    listener: UnixListener,
}

impl IpcServer {
    pub fn bind() -> Result<Self, std::io::Error> {
        let path = get_socket_path();
        if path.exists() {
            let _ = std::fs::remove_file(&path);
        }
        let listener = UnixListener::bind(path)?;
        Ok(Self { listener })
    }

    pub async fn accept(&self) -> Result<(UnixStream, IpcCommand), Box<dyn std::error::Error + Send + Sync>> {
        let (mut stream, _) = self.listener.accept().await?;
        let mut buffer = Vec::new();
        stream.read_to_end(&mut buffer).await?;
        let command: IpcCommand = serde_json::from_slice(&buffer)?;
        Ok((stream, command))
    }
}

    pub async fn collect_cpu() -> CpuMetrics {
        let mut temp_c = 0;
        for i in 0..10 {
            let path = format!("/sys/class/thermal/thermal_zone{}/temp", i);
            if let Ok(raw) = fs::read_to_string(&path) {
                if let Ok(v) = raw.trim().parse::<u32>() {
                    temp_c = if v > 1000 { v / 1000 } else { v };
                    if temp_c > 0 {
                        break;
                    }
                }
            }
        }

        let mut usage_pct = 0;
        if let Ok(loadavg) = fs::read_to_string("/proc/loadavg") {
            if let Some(first) = loadavg.split_whitespace().next() {
                if let Ok(val) = first.parse::<f32>() {
                    let cores = std::thread::available_parallelism()
                        .map(|c| c.get())
                        .unwrap_or(1) as f32;
                    usage_pct = ((val / cores) * 100.0).clamp(0.0, 100.0) as u8;
                }
            }
        }

        let top_procs = Self::get_top_cpu_processes().await;

        CpuMetrics {
            usage_pct,
            temp_c,
            top_procs,
        }
    }

    pub async fn collect_gpu(card_id: &str, is_nvidia: bool, nv_index: u32) -> GpuMetrics {
        if is_nvidia {
            if let Ok(output) = Command::new("nvidia-smi")
                .args([
                    &format!("--id={}", nv_index),
                    "--query-gpu=utilization.gpu,temperature.gpu,power.draw,memory.free",
                    "--format=csv,noheader,nounits",
                ])
                .output()
                .await
            {
                let text = String::from_utf8_lossy(&output.stdout);
                let parts: Vec<&str> = text.split(',').map(|s| s.trim()).collect();
                if parts.len() >= 4 {
                    let u = parts[0].parse().unwrap_or(0);
                    let t = parts[1].parse().unwrap_or(0);
                    let p = parts[2].parse::<f32>().unwrap_or(0.0) as u32;
                    let v = parts[3].parse::<u32>().unwrap_or(0) / 1024;
                    return GpuMetrics {
                        name: format!("NVIDIA GPU {}", nv_index),
                        usage_pct: u,
                        temp_c: t,
                        power_w: p,
                        vram_free_gib: v,
                        top_procs: Vec::new(),
                    };
                }
            }
        }

        let base = format!("/sys/class/drm/{}/device", card_id);
        let usage_pct = fs::read_to_string(format!("{}/gpu_busy_percent", base))
            .ok()
            .and_then(|s| s.trim().parse::<u8>().ok())
            .unwrap_or(0);

        let temp_c = fs::read_to_string(format!("{}/hwmon/hwmon0/temp1_input", base))
            .ok()
            .and_then(|s| s.trim().parse::<u32>().ok())
            .map(|t| t / 1000)
            .unwrap_or(0);

        let power_w = fs::read_to_string(format!("{}/hwmon/hwmon0/power1_average", base))
            .ok()
            .and_then(|s| s.trim().parse::<u32>().ok())
            .map(|p| p / 1_000_000)
            .unwrap_or(0);

        let vram_free_gib = (|| {
            let total = fs::read_to_string(format!("{}/mem_info_vram_total", base))
                .ok()?
                .trim()
                .parse::<u64>()
                .ok()?;
            let used = fs::read_to_string(format!("{}/mem_info_vram_used", base))
                .ok()?
                .trim()
                .parse::<u64>()
                .ok()?;
            Some(((total.saturating_sub(used)) / (1024 * 1024 * 1024)) as u32)
        })()
        .unwrap_or(0);

        GpuMetrics {
            name: card_id.to_uppercase(),
            usage_pct,
            temp_c,
            power_w,
            vram_free_gib,
            top_procs: Vec::new(),
        }
    }

    pub async fn collect_ram() -> RamMetrics {
        let meminfo = fs::read_to_string("/proc/meminfo").unwrap_or_default();
        let mut total_kb = 0.0;
        let mut avail_kb = 0.0;

        for line in meminfo.lines() {
            if line.starts_with("MemTotal:") {
                total_kb = line.split_whitespace().nth(1).and_then(|s| s.parse().ok()).unwrap_or(0.0);
            } else if line.starts_with("MemAvailable:") {
                avail_kb = line.split_whitespace().nth(1).and_then(|s| s.parse().ok()).unwrap_or(0.0);
            }
        }

        let total_gib = total_kb / (1024.0 * 1024.0);
        let avail_gib = avail_kb / (1024.0 * 1024.0);

        let zram_orig = fs::read_to_string("/sys/block/zram0/orig_data_size")
            .ok()
            .and_then(|s| s.trim().parse::<f32>().ok())
            .unwrap_or(0.0);
        let zram_comp = fs::read_to_string("/sys/block/zram0/compr_data_size")
            .ok()
            .and_then(|s| s.trim().parse::<f32>().ok())
            .unwrap_or(0.0);

        let mut zram_ratio = 1.0;
        let mut zram_saved_gib = 0.0;
        if zram_comp > 0.0 && zram_orig > 0.0 {
            zram_ratio = zram_orig / zram_comp;
            zram_saved_gib = (zram_orig - zram_comp) / (1024.0 * 1024.0 * 1024.0);
        }

        let top_procs = Self::get_top_ram_processes().await;

        RamMetrics {
            total_gib,
            available_gib: avail_gib,
            effective_avail_gib: avail_gib + zram_saved_gib,
            zram_ratio,
            zram_saved_gib,
            top_procs,
        }
    }

    pub fn collect_battery() -> BatteryMetrics {
        let ps = Path::new("/sys/class/power_supply");
        if !ps.exists() {
            return BatteryMetrics::default();
        }

        let mut bat_dir = None;
        if let Ok(entries) = fs::read_dir(ps) {
            for entry in entries.flatten() {
                let name = entry.file_name().to_string_lossy().to_string();
                if name.starts_with("BAT") {
                    bat_dir = Some(entry.path());
                    break;
                }
            }
        }

        let Some(b) = bat_dir else {
            return BatteryMetrics {
                has_battery: false,
                percent: 100,
                status: "Mains".into(),
                power_w: 0.0,
            };
        };

        let cap = fs::read_to_string(b.join("capacity"))
            .ok()
            .and_then(|s| s.trim().parse::<u8>().ok())
            .unwrap_or(0);

        let status = fs::read_to_string(b.join("status"))
            .unwrap_or_else(|_| "Unknown".into())
            .trim()
            .to_string();

        let power_now = fs::read_to_string(b.join("power_now"))
            .ok()
            .and_then(|s| s.trim().parse::<f32>().ok())
            .map(|p| p / 1_000_000.0)
            .unwrap_or(0.0);

        BatteryMetrics {
            has_battery: true,
            percent: cap,
            status,
            power_w: power_now,
        }
    }

    pub async fn update_net(&mut self) -> NetMetrics {
        let dev = fs::read_to_string("/proc/net/dev").unwrap_or_default();
        let mut best_iface = "none".to_string();
        let mut best_rx: u64 = 0;
        let mut best_tx: u64 = 0;

        for line in dev.lines() {
            let parts: Vec<&str> = line.split_whitespace().collect();
            if parts.len() >= 10 {
                let iface = parts[0].trim_end_matches(':');
                if iface != "lo" {
                    let rx = parts[1].parse::<u64>().unwrap_or(0);
                    let tx = parts[9].parse::<u64>().unwrap_or(0);
                    if rx > best_rx {
                        best_rx = rx;
                        best_tx = tx;
                        best_iface = iface.to_string();
                    }
                }
            }
        }

        let now = std::time::Instant::now();
        let dt = now.duration_since(self.prev_net_time).as_secs_f32().max(0.1);
        let rx_rate = (best_rx.saturating_sub(self.prev_rx_bytes) as f32 / dt) * 8.0;
        let tx_rate = (best_tx.saturating_sub(self.prev_tx_bytes) as f32 / dt) * 8.0;

        self.prev_rx_bytes = best_rx;
        self.prev_tx_bytes = best_tx;
        self.prev_net_time = now;

        let hw_type = if best_iface.starts_with("wl") {
            "WiFi"
        } else if best_iface.starts_with("ww") {
            "Cellular"
        } else {
            "Ethernet"
        };

        NetMetrics {
            iface: best_iface,
            hw_type: hw_type.to_string(),
            ip: "".into(),
            down_str: Self::format_bits(rx_rate),
            up_str: Self::format_bits(tx_rate),
            ping_str: "--".into(),
        }
    }

    fn format_bits(bps: f32) -> String {
        if bps >= 1_000_000_000.0 {
            format!("{:.1}Gb", bps / 1_000_000_000.0)
        } else if bps >= 1_000_000.0 {
            format!("{:.1}Mb", bps / 1_000_000.0)
        } else {
            format!("{:.1}Kb", (bps / 1_000.0).max(0.0))
        }
    }

    async fn get_top_cpu_processes() -> Vec<ProcessItem> {
        let Ok(out) = Command::new("ps")
            .args(["-eo", "pid,pcpu,comm", "--sort=-pcpu", "--no-headers"])
            .output()
            .await
        else {
            return Vec::new();
        };

        String::from_utf8_lossy(&out.stdout)
            .lines()
            .take(8)
            .filter_map(|l| {
                let parts: Vec<&str> = l.split_whitespace().collect();
                if parts.len() >= 3 {
                    let pid = parts[0].parse().ok()?;
                    let cpu = parts[1];
                    let name = parts[2..].join(" ");
                    Some(ProcessItem {
                        pid,
                        name,
                        metric: format!("{}%", cpu),
                        is_highlight: false,
                    })
                } else {
                    None
                }
            })
            .collect()
    }

    async fn get_top_ram_processes() -> Vec<ProcessItem> {
        let Ok(out) = Command::new("ps")
            .args(["-eo", "pid,rss,comm", "--sort=-rss", "--no-headers"])
            .output()
            .await
        else {
            return Vec::new();
        };

        String::from_utf8_lossy(&out.stdout)
            .lines()
            .take(8)
            .filter_map(|l| {
                let parts: Vec<&str> = l.split_whitespace().collect();
                if parts.len() >= 3 {
                    let pid = parts[0].parse().ok()?;
                    let rss_kb: u64 = parts[1].parse().ok()?;
                    let name = parts[2..].join(" ");
                    Some(ProcessItem {
                        pid,
                        name,
                        metric: format!("{} MiB", rss_kb / 1024),
                        is_highlight: false,
                    })
                } else {
                    None
                }
            })
            .collect()
    }

    pub async fn kill_process(pid: u32) -> Result<(), std::io::Error> {
        if pid <= 1 {
            return Err(std::io::Error::new(
                std::io::ErrorKind::PermissionDenied,
                "Refusing to terminate PID <= 1",
            ));
        }
        Command::new("kill").args(["-9", &pid.to_string()]).status().await?;
        Ok(())
    }

use tokio::process::Command;

#[derive(Debug, Clone, Default)]
pub struct AudioStatus {
    pub volume_pct: u8,
    pub is_muted: bool,
}

pub struct AudioBackend;

impl AudioBackend {
    pub async fn get_sink() -> AudioStatus {
        if let Ok(out) = Command::new("wpctl")
            .args(["get-volume", "@DEFAULT_AUDIO_SINK@"])
            .output()
            .await
        {
            let text = String::from_utf8_lossy(&out.stdout);
            return Self::parse_wpctl(&text);
        }

        if let Ok(out) = Command::new("pactl")
            .args(["get-sink-volume", "@DEFAULT_SINK@"])
            .output()
            .await
        {
            let text = String::from_utf8_lossy(&out.stdout);
            let mute_out = Command::new("pactl")
                .args(["get-sink-mute", "@DEFAULT_SINK@"])
                .output()
                .await
                .map(|o| String::from_utf8_lossy(&o.stdout).contains("yes"))
                .unwrap_or(false);

            if let Some(pos) = text.find('%') {
                if let Some(start) = text[..pos].rfind(|c: char| !c.is_ascii_digit()) {
                    let vol = text[start + 1..pos].parse().unwrap_or(0);
                    return AudioStatus {
                        volume_pct: vol,
                        is_muted: mute_out,
                    };
                }
            }
        }

        AudioStatus::default()
    }

    pub async fn get_source() -> AudioStatus {
        if let Ok(out) = Command::new("wpctl")
            .args(["get-volume", "@DEFAULT_AUDIO_SOURCE@"])
            .output()
            .await
        {
            let text = String::from_utf8_lossy(&out.stdout);
            return Self::parse_wpctl(&text);
        }
        AudioStatus::default()
    }

    pub async fn adjust_sink_volume(step_pct: i8) {
        let sign = if step_pct >= 0 { "+" } else { "-" };
        let frac = format!("{:.2}{}", (step_pct.abs() as f32) / 100.0, sign);
        let _ = Command::new("wpctl")
            .args(["set-volume", "@DEFAULT_AUDIO_SINK@", &frac])
            .status()
            .await;
    }

    pub async fn toggle_sink_mute() {
        let _ = Command::new("wpctl")
            .args(["set-mute", "@DEFAULT_AUDIO_SINK@", "toggle"])
            .status()
            .await;
    }

    pub async fn toggle_source_mute() {
        let _ = Command::new("wpctl")
            .args(["set-mute", "@DEFAULT_AUDIO_SOURCE@", "toggle"])
            .status()
            .await;
    }

    pub async fn cycle_sinks() {
        if let Ok(out) = Command::new("wpctl").arg("status").output().await {
            let text = String::from_utf8_lossy(&out.stdout);
            let mut sinks = Vec::new();
            let mut in_sinks = false;
            let mut current_id = None;

            for line in text.lines() {
                if line.contains("Sinks:") {
                    in_sinks = true;
                } else if line.contains("Sources:") || line.contains("Filters:") {
                    in_sinks = false;
                } else if in_sinks {
                    if let Some(num_str) = line.split('.').next() {
                        let id_str = num_str.trim().trim_start_matches('*').trim();
                        if let Ok(id) = id_str.parse::<u32>() {
                            sinks.push(id);
                            if line.contains('*') {
                                current_id = Some(id);
                            }
                        }
                    }
                }
            }

            if sinks.len() > 1 {
                let next_id = if let Some(cur) = current_id {
                    let idx = sinks.iter().position(|&x| x == cur).unwrap_or(0);
                    sinks[(idx + 1) % sinks.len()]
                } else {
                    sinks[0]
                };

                let _ = Command::new("wpctl")
                    .args(["set-default", &next_id.to_string()])
                    .status()
                    .await;
            }
        }
    }

    fn parse_wpctl(text: &str) -> AudioStatus {
        let is_muted = text.contains("[MUTED]");
        let mut volume_pct = 0;
        if let Some(vol_str) = text.strip_prefix("Volume:") {
            let num_part = vol_str.split_whitespace().next().unwrap_or("0.0");
            if let Ok(f) = num_part.parse::<f32>() {
                volume_pct = (f * 100.0).round().clamp(0.0, 150.0) as u8;
            }
        }
        AudioStatus {
            volume_pct,
            is_muted,
        }

pub struct WeatherData {

    pub desc: &'static str,

#[derive(Deserialize)]
struct OpenMeteoResponse {
    current: Option<CurrentMeteo>,
    hourly: Option<HourlyMeteo>,
}

#[derive(Deserialize)]
#[allow(dead_code)]
struct CurrentMeteo {
    temperature_2m: Option<f32>,
    weather_code: Option<u16>,
    wind_speed_10m: Option<f32>,
}

#[derive(Deserialize)]
struct HourlyMeteo {
    time: Vec<String>,
    temperature_2m: Vec<f32>,
    weather_code: Vec<u16>,
    precipitation_probability: Option<Vec<u8>>,
    wind_speed_10m: Option<Vec<f32>>,
}

pub struct WeatherBackend;

impl WeatherBackend {
    pub async fn fetch() -> Result<WeatherData, Box<dyn std::error::Error>> {
        let (lat, lon) = Self::detect_coords().await;
        let url = format!(
            "https://api.open-meteo.com/v1/forecast?latitude={}&longitude={}&current=temperature_2m,weather_code,wind_speed_10m&hourly=temperature_2m,precipitation_probability,weather_code,wind_speed_10m&forecast_days=2&timezone=auto",
            lat, lon
        );

        let output = Command::new("curl")
            .args(["-s", "--max-time", "6", &url])
            .output()
            .await?;

        let res: OpenMeteoResponse = serde_json::from_slice(&output.stdout)?;
        let cur = res.current.ok_or("No current weather data")?;
        let hourly = res.hourly.ok_or("No hourly data")?;

        let cur_c = cur.temperature_2m.unwrap_or(20.0).round() as i32;
        let cur_f = (cur_c as f32 * 9.0 / 5.0 + 32.0).round() as i32;
        let (cur_desc, cur_icon) = Self::wmo_info(cur.weather_code.unwrap_or(0));

        let mut alert = WeatherAlert::None;
        let mut timeline = Vec::new();
        let total = hourly.time.len().min(24);

        for i in 0..total {
            let tc = hourly.temperature_2m.get(i).copied().unwrap_or(20.0).round() as i32;
            let tf = (tc as f32 * 9.0 / 5.0 + 32.0).round() as i32;
            let code = hourly.weather_code.get(i).copied().unwrap_or(0);
            let (desc, icon) = Self::wmo_info(code);
            let prob = hourly.precipitation_probability.as_ref().and_then(|p| p.get(i).copied()).unwrap_or(0);
            let wind = hourly.wind_speed_10m.as_ref().and_then(|w| w.get(i).copied()).unwrap_or(0.0);

            let is_danger = tf >= 95 || tf <= 30 || code >= 95 || wind >= 60.0;
            if is_danger && i <= 3 && alert == WeatherAlert::None {
                alert = WeatherAlert::Danger(format!("Severe Weather ({}h)", i));
            } else if prob >= 60 && i <= 3 && alert == WeatherAlert::None {
                alert = WeatherAlert::Upcoming(format!("Rain in {}h", i));
            }

            let hour_raw = hourly.time[i].split('T').nth(1).unwrap_or("12:00");
            timeline.push(HourlyForecast {
                time_label: hour_raw.to_string(),
                icon,
                temp_f: tf,
                temp_c: tc,
                desc,
                precip_prob: prob,
                is_danger,
            });
        }

        Ok(WeatherData {
            temp_f: cur_f,
            temp_c: cur_c,
            icon: cur_icon,
            desc: cur_desc,
            alert,
            hourly: timeline,
        })
    }

    async fn detect_coords() -> (String, String) {
        if let Ok(out) = Command::new("curl")
            .args(["-s", "--max-time", "3", "http://ip-api.com/json"])
            .output()
            .await
        {
            if let Ok(val) = serde_json::from_slice::<serde_json::Value>(&out.stdout) {
                if let (Some(lat), Some(lon)) = (val["lat"].as_f64(), val["lon"].as_f64()) {
                    return (lat.to_string(), lon.to_string());
                }
            }
        }
        ("40.7128".into(), "-74.0060".into())
    }

    fn wmo_info(code: u16) -> (&'static str, &'static str) {
        match code {
            0 => ("Clear", "☀️"),
            1 => ("Mainly Clear", "🌤️"),
            2 => ("Partly Cloudy", "⛅"),
            3 => ("Overcast", "☁️"),
            45 | 48 => ("Fog", "🌫️"),
            51..=55 => ("Drizzle", "🌦️"),
            61..=65 => ("Rain", "🌧️"),
            71..=77 => ("Snow", "❄️"),
            80..=82 => ("Showers", "⛈️"),
            95..=99 => ("Thunderstorm", "⛈️"),
            _ => ("Fair", "🌤️"),
        }
    }

impl ClockBackend {
    pub fn get_bar_time() -> String {
        Local::now().format("%I:%M %p").to_string()
    }

    pub fn get_world_matrix() -> Vec<WorldZoneColumn> {
        let now_utc = Utc::now();
        let local_code = Local::now().format("%Z").to_string();

        let americas = vec![
            ("Pacific Time", "US/PT", -7.0),
            ("Mountain Time", "US/MT", -6.0),
            ("Central Time", "US/CT", -5.0),
            ("Eastern Time", "US/ET", -4.0),
            ("São Paulo", "BRT", -3.0),
        ];

        let europe = vec![
            ("UTC / GMT", "UTC", 0.0),
            ("London", "BST", 1.0),
            ("Berlin / Paris", "CEST", 2.0),
            ("Cairo", "EEST", 3.0),
            ("Moscow", "MSK", 3.0),
        ];

        let asia = vec![
            ("Dubai", "GST", 4.0),
            ("India", "IST", 5.5),
            ("Singapore", "SGT", 8.0),
            ("Tokyo", "JST", 9.0),
            ("Sydney", "AEST", 10.0),
        ];

        let render_column = |title: &'static str, zones: Vec<(&'static str, &'static str, f32)>| {
            let rendered = zones
                .into_iter()
                .map(|(name, code, offset)| {
                    let zone_time = now_utc + chrono::Duration::minutes((offset * 60.0) as i64);
                    let formatted = zone_time.format("%I:%M %p").to_string();
                    let is_me = code == local_code;
                    (name, code, formatted, is_me)
                })
                .collect();

            WorldZoneColumn { title, zones: rendered }
        };

        vec![
            render_column("AMERICAS", americas),
            render_column("EUROPE & AFRICA", europe),
            render_column("ASIA & PACIFIC", asia),
        ]
    }

    pub fn compute_calendar(year: i32, month: u32) -> CalendarViewData {
        let now = Local::now();
        let y = if year > 0 { year } else { now.year() };
        let m = if month >= 1 && month <= 12 { month } else { now.month() };

        let first_day = NaiveDate::from_ymd_opt(y, m, 1).unwrap_or_default();
        let first_day_offset = first_day.weekday().num_days_from_sunday() as usize;

        let days_in_month = if m == 12 {
            NaiveDate::from_ymd_opt(y + 1, 1, 1)
        } else {
            NaiveDate::from_ymd_opt(y, m + 1, 1)
        }
        .unwrap_or_default()
        .signed_duration_since(first_day)
        .num_days() as u32;

        let today_day = if y == now.year() && m == now.month() {
            now.day()
        } else {
            0
        };

        CalendarViewData {
            date_str: now.format("%a, %b %d").to_string(),
            month_label: first_day.format("%B %Y").to_string(),
            year: y,
            month: m,
            first_day_offset,
            days_in_month,
            today_day,
        }
    }

pub struct PodmanBackend;

impl PodmanBackend {
    pub async fn poll() -> PodmanStatus {
        let engine = if Command::new("podman").arg("version").output().await.is_ok() {
            "podman"
        } else {
            "docker"
        };

        let Ok(output) = Command::new(engine)
            .args(["ps", "-a", "--format", "{{.Names}}\t{{.Status}}\t{{.Image}}\t{{.State}}"])
            .output()
            .await
        else {
            return PodmanStatus {
                summary_label: "Podman Offline".into(),
                ..Default::default()
            };
        };

        let raw = String::from_utf8_lossy(&output.stdout);
        let mut containers = Vec::new();
        let mut watchers = Vec::new();
        let mut any_limited = false;

        for line in raw.lines() {
            let parts: Vec<&str> = line.split('\t').collect();
            if parts.is_empty() || parts[0].trim().is_empty() {
                continue;
            }

            let name = parts[0].trim().to_string();
            let status_raw = parts.get(1).copied().unwrap_or("Unknown");
            let image = parts.get(2).copied().unwrap_or("");
            let state = parts.get(3).copied().unwrap_or("").to_lowercase();
            let is_running = state == "running" || status_raw.to_lowercase().starts_with("up");

            let is_watcher = name.contains("twitch") || name.contains("miner") || name.contains("berry");
            let mut watcher_details = WatcherDetails {
                name: name.clone(),
                is_running,
                ..Default::default()
            };

            if is_watcher && is_running {
                let (streamer, drops, game, limited) = Self::parse_logs(engine, &name).await;
                watcher_details.streamer = streamer;
                watcher_details.drops = drops;
                watcher_details.game = game;
                watcher_details.is_limited = limited;
                if limited {
                    any_limited = true;
                }
                watchers.push(watcher_details);
            }

            let purpose = Self::determine_purpose(&name, image);
            containers.push(ContainerInfo {
                name,
                status: status_raw.to_string(),
                is_running,
                is_watcher,
                purpose,
            });
        }

        let summary_label = if any_limited {
            "Miner: Limited!".to_string()
        } else if let Some(w) = watchers.iter().find(|w| w.is_running && !w.streamer.is_empty()) {
            format!("📺 {}", w.streamer)
        } else {
            let active_count = containers.iter().filter(|c| c.is_running).count();
            if active_count > 0 {
                format!("{} Active", active_count)
            } else {
                "Idle".into()
            }
        };

        PodmanStatus {
            is_ok: !containers.is_empty(),
            is_rate_limited: any_limited,
            watchers,
            containers,
            summary_label,
        }
    }

    async fn parse_logs(engine: &str, container_name: &str) -> (String, String, String, bool) {
        let Ok(out) = Command::new(engine)
            .args(["logs", "--tail", "60", container_name])
            .output()
            .await
        else {
            return (String::new(), String::new(), String::new(), false);
        };

        let logs = String::from_utf8_lossy(&out.stdout);
        let mut streamer = String::new();
        let mut drops = String::new();
        let mut game = String::new();
        let is_limited = logs.contains("429") || logs.contains("rate limit") || logs.contains("too many requests");

        for line in logs.lines().rev() {
            if streamer.is_empty() {
                if let Some(pos) = line.find("Watching:") {
                    streamer = line[pos + 9..].split_whitespace().next().unwrap_or("").to_string();
                } else if let Some(pos) = line.find("Channel:") {
                    streamer = line[pos + 8..].split_whitespace().next().unwrap_or("").to_string();
                }
            }

            if drops.is_empty() && line.contains("drop") {
                if let Some(pos) = line.find("Drop progress:") {
                    drops = line[pos..].trim().to_string();
                }
            }

            if game.is_empty() && line.contains("game:") {
                if let Some(start) = line.find("game:") {
                    let rest = &line[start + 5..];
                    if let Some(end) = rest.find(')') {
                        game = rest[..end].trim().to_string();
                    }
                }
            }

            if !streamer.is_empty() && !drops.is_empty() {
                break;
            }
        }

        (streamer, drops, game, is_limited)
    }

    fn determine_purpose(name: &str, img: &str) -> String {

        if low.contains("flaresolverr") {

        } else if low.contains("jellyfin") {
            "Media Server (:8096)".into()
        } else if low.contains("vaultwarden") {
            "Bitwarden Vault".into()
        } else {
            "Background Service".into()
        }
    }

#[derive(Deserialize)]
#[allow(dead_code)]
struct RawBorgPayload {
    status: Option<String>,
    percent: Option<u8>,
    archive_name: Option<String>,
    speed: Option<String>,
    eta: Option<String>,
    processed_files: Option<u64>,
    total_files: Option<u64>,
}

pub struct BorgBackend;

impl BorgBackend {
    pub async fn poll() -> BorgStatus {
        let mut payload_raw = None;
        let candidates = [
            PathBuf::from("/dev/shm/borg-offsite-status.json"),
            dirs::runtime_dir()
                .unwrap_or_else(|| PathBuf::from("/tmp"))
                .join("borg-offsite-status.json"),
        ];

        for path in &candidates {
            if let Ok(content) = fs::read_to_string(path) {
                if !content.trim().is_empty() {
                    payload_raw = Some(content);
                    break;
                }
            }
        }

        let is_mounted = fs::read_to_string("/proc/mounts")
            .map(|s| s.contains("/tmp/borg-mount") || s.contains("/mnt/main_backup"))
            .unwrap_or(false);

        let Some(raw) = payload_raw else {
            return BorgStatus {
                state_label: if is_mounted { "Mounted".into() } else { "Idle".into() },
                is_active: false,
                is_mounted,
                percent: 0,
                speed: "".into(),
                eta: "".into(),
                details: vec![if is_mounted {
                    "Archive Mounted at /tmp/borg-mount".into()
                } else {
                    "Standing by (All repositories synced)".into()
                }],
            };
        };

        let parsed: RawBorgPayload = serde_json::from_str(&raw).unwrap_or(RawBorgPayload {
            status: Some("idle".into()),
            percent: None,
            archive_name: None,
            speed: None,
            eta: None,
            processed_files: None,
            total_files: None,
        });

        let st = parsed.status.unwrap_or_else(|| "idle".into());
        let pct = parsed.percent.unwrap_or(0);
        let speed = parsed.speed.unwrap_or_default();
        let eta = parsed.eta.unwrap_or_default();

        let mut details = Vec::new();
        let is_active = st == "running" || st == "syncing" || st == "indexing";

        match st.as_str() {
            "indexing" => {
                details.push("Indexing repository files...".into());
            }
            "running" => {
                details.push(format!("Active Backup Progress: {}%", pct));
                if let (Some(done), Some(tot)) = (parsed.processed_files, parsed.total_files) {
                    details.push(format!("Files: {} / {}", done, tot));
                }
            }
            "syncing" => {
                details.push(format!("Syncing to WebDAV: {}%", pct));
            }
            _ => {
                details.push("Repository Idle".into());
            }
        }

        if !speed.is_empty() {
            details.push(format!("Transfer Speed: {}", speed));
        }
        if !eta.is_empty() {
            details.push(format!("ETA: {}", eta));
        }

        let state_label = if is_active {
            if pct > 0 {
                format!("Borg: {}%", pct)
            } else {
                "Borg: Active".into()
            }
        } else if is_mounted {
            "Borg: Mounted".into()
        } else {
            "Borg: Idle".into()
        };

        BorgStatus {
            state_label,
            is_active,
            is_mounted,
            percent: pct,
            speed,
            eta,
            details,
        }
    }

#[derive(Debug, Clone)]
pub enum CalcMessage {
    InputChanged(String),
    CopyResult(String),
    Dismiss,
}

pub struct CalculatorOverlay {
    pub input_text: String,
    pub result_text: String,
}

impl CalculatorOverlay {
    pub fn new() -> Self {
        Self {
            input_text: String::new(),
            result_text: String::new(),
        }
    }

    pub fn update(&mut self, message: CalcMessage) {
        if let CalcMessage::InputChanged(val) = message {
            self.input_text = val.clone();
            self.result_text = Self::evaluate(&val);
        }
    }

    pub fn evaluate(expr: &str) -> String {
        let clean = expr.trim().trim_start_matches('=').trim();
        if clean.is_empty() {
            return String::new();
        }

        if let Some(res) = Self::eval_units(clean) {
            return res;
        }

        match Self::eval_math(clean) {
            Ok(v) => format!("{:.6}", v)
                .trim_end_matches('0')
                .trim_end_matches('.')
                .to_string(),
            Err(_) => "Syntax Error".to_string(),
        }
    }

    fn eval_math(expr: &str) -> Result<f64, ()> {
        let mut tokens = Vec::new();
        let mut num_buf = String::new();

        for c in expr.chars() {
            if c.is_ascii_digit() || c == '.' {
                num_buf.push(c);
            } else {
                if !num_buf.is_empty() {
                    tokens.push(num_buf.clone());
                    num_buf.clear();
                }
                if !c.is_whitespace() {
                    tokens.push(c.to_string());
                }
            }
        }
        if !num_buf.is_empty() {
            tokens.push(num_buf);
        }

        let mut it = tokens.iter().peekable();
        Self::parse_expr(&mut it)
    }

    fn parse_expr<'a>(it: &mut std::iter::Peekable<std::slice::Iter<'a, String>>) -> Result<f64, ()> {
        let mut left = Self::parse_term(it)?;
        while let Some(&op) = it.peek() {
            if op == "+" {
                it.next();
                left += Self::parse_term(it)?;
            } else if op == "-" {
                it.next();
                left -= Self::parse_term(it)?;
            } else {
                break;
            }
        }
        Ok(left)
    }

    fn parse_term<'a>(it: &mut std::iter::Peekable<std::slice::Iter<'a, String>>) -> Result<f64, ()> {
        let mut left = Self::parse_factor(it)?;
        while let Some(&op) = it.peek() {
            if op == "*" {
                it.next();
                left *= Self::parse_factor(it)?;
            } else if op == "/" {
                it.next();
                let denom = Self::parse_factor(it)?;
                if denom == 0.0 {
                    return Err(());
                }
                left /= denom;
            } else {
                break;
            }
        }
        Ok(left)
    }

    fn parse_factor<'a>(it: &mut std::iter::Peekable<std::slice::Iter<'a, String>>) -> Result<f64, ()> {
        let token = it.next().ok_or(())?;
        if token == "(" {
            let val = Self::parse_expr(it)?;
            if it.next().map(|s| s.as_str()) != Some(")") {
                return Err(());
            }
            Ok(val)
        } else if token == "-" {
            Ok(-Self::parse_factor(it)?)
        } else {
            token.parse::<f64>().map_err(|_| ())
        }
    }

    fn eval_units(expr: &str) -> Option<String> {
        let parts: Vec<&str> = expr.split_whitespace().collect();
        if parts.len() == 4 && (parts[2].eq_ignore_ascii_case("in") || parts[2].eq_ignore_ascii_case("to")) {
            let val = parts[0].parse::<f64>().ok()?;
            let from = parts[1].to_lowercase();
            let to = parts[3].to_lowercase();

            if from == "c" && to == "f" {
                return Some(format!("{:.2} °F", val * 9.0 / 5.0 + 32.0));
            } else if from == "f" && to == "c" {
                return Some(format!("{:.2} °C", (val - 32.0) * 5.0 / 9.0));
            } else if from == "kg" && to == "lbs" {
                return Some(format!("{:.2} lbs", val * 2.20462));
            } else if from == "lbs" && to == "kg" {
                return Some(format!("{:.2} kg", val / 2.20462));
            } else if from == "km" && to == "mi" {
                return Some(format!("{:.2} mi", val * 0.621371));
            } else if from == "mi" && to == "km" {
                return Some(format!("{:.2} km", val / 0.621371));
            }
        }
        None
    }

    pub async fn copy_to_clipboard(val: &str) {
        if val.is_empty() {
            return;
        }
        let _ = Command::new("wl-copy").arg(val).status().await;
        let _ = Command::new("notify-send")
            .args(["-a", "Calculator", "Copied to Clipboard", val])
            .spawn();
    }

        let (win_w, win_h) = settings.get_window_size("calc", 820, 580);

    pub async fn reload() -> Vec<ClipItem> {
        let Ok(out) = Command::new("cliphist").arg("list").output().await else {
            return Vec::new();
        };

        let raw = String::from_utf8_lossy(&out.stdout);
        let mut list = Vec::new();

        for line in raw.lines() {
            if let Some((id_str, content)) = line.split_once('\t') {
                if let Ok(id) = id_str.trim().parse::<u64>() {
                    let is_image = content.contains("[[ binary data") || content.contains("binary data");
                    list.push(ClipItem {
                        id,
                        text: content.to_string(),
                        is_image,
                    });
                }
            }
        }
        list
    }

    pub async fn copy(id: u64) {
        if let Ok(decode) = Command::new("cliphist")
            .args(["decode", &id.to_string()])
            .output()
            .await
        {
            let child = Command::new("wl-copy")
                .stdin(std::process::Stdio::piped())
                .spawn()
                .ok();
            if let Some(mut c) = child {
                use tokio::io::AsyncWriteExt;
                if let Some(mut stdin) = c.stdin.take() {
                    let _ = stdin.write_all(&decode.stdout).await;
                }
            }
        }
    }

    pub async fn delete(id: u64) {
        let child = Command::new("cliphist")
            .arg("delete")
            .stdin(std::process::Stdio::piped())
            .spawn()
            .ok();
        if let Some(mut c) = child {
            use tokio::io::AsyncWriteExt;
            if let Some(mut stdin) = c.stdin.take() {
                let _ = stdin.write_all(format!("{}\t\n", id).as_bytes()).await;
            }
        }
    }

    pub fn save(&self) {
        let path = Self::storage_path();
        if let Some(parent) = path.parent() {
            let _ = fs::create_dir_all(parent);
        }
        if let Ok(data) = serde_json::to_string_pretty(&self.notes) {
            let tmp = path.with_extension("tmp");
            if fs::write(&tmp, data).is_ok() {
                let _ = fs::rename(tmp, path);
            }
        }
    }

            NotesMessage::SearchOrNewChanged(val) => {
                self.input_text = val;

    pub fn scan_passwords() -> Vec<String> {
        let store_dir = std::env::var("PASSWORD_STORE_DIR")
            .map(PathBuf::from)
            .unwrap_or_else(|_| {
                dirs::home_dir()
                    .unwrap_or_else(|| PathBuf::from("/tmp"))
                    .join(".password-store")
            });

        let mut keys = Vec::new();
        if !store_dir.exists() {
            return keys;
        }

        for entry in WalkDir::new(&store_dir).into_iter().flatten() {
            let path = entry.path();
            if path.is_file() && path.extension().and_then(|s| s.to_str()) == Some("gpg") {
                if let Ok(rel) = path.strip_prefix(&store_dir) {
                    let mut s = rel.to_string_lossy().to_string();
                    if s.ends_with(".gpg") {
                        s.truncate(s.len() - 4);
                        if !s.starts_with(".git") {
                            keys.push(s);
                        }
                    }
                }
            }
        }

        keys.sort();
        keys
    }

            self.search_query = q.clone();
            let lower = q.trim().to_lowercase();

                self.filtered_keys = self.all_keys.clone();

                self.filtered_keys = self
                    .all_keys
                    .iter()
                    .filter(|k| k.to_lowercase().contains(&lower))

                    .collect();
            }
        }
    }

        match message {
            PowerMessage::Select(act) => {
                match act {
                    PowerAction::Lock | PowerAction::Sleep => {
                        Self::execute(act);
                    }
                    PowerAction::Logout | PowerAction::Reboot | PowerAction::Shutdown => {
                        if self.confirming == Some(act) {
                            Self::execute(act);
                            self.confirming = None;
                        } else {
                            self.confirming = Some(act);
                        }
                    }
                }
            }
            PowerMessage::Confirm(act) => {
                Self::execute(act);
                self.confirming = None;
            }
            PowerMessage::Dismiss => {
                self.confirming = None;
            }
        }
    }

    pub fn execute(action: PowerAction) {
        tokio::spawn(async move {
            match action {
                PowerAction::Lock => {
                    // Triggers session lockscreen
                }
                PowerAction::Sleep => {
                    let _ = Command::new("systemctl").arg("suspend").status().await;
                }
                PowerAction::Reboot => {
                    let _ = Command::new("systemctl").arg("reboot").status().await;
                }
                PowerAction::Shutdown => {
                    let _ = Command::new("systemctl").arg("poweroff").status().await;
                }
                PowerAction::Logout => {
                    if std::env::var("HYPRLAND_INSTANCE_SIGNATURE").is_ok() {
                        let _ = Command::new("hyprctl").args(["dispatch", "exit"]).status().await;
                    } else if std::env::var("SWAYSOCK").is_ok() {

                        let _ = Command::new("loginctl").args(["terminate-session", "self"]).status().await;

        });

    pub fn parse_when(when_expr: &str) -> (i64, String) {
        let now = Local::now();
        let expr = when_expr.trim().to_lowercase();

        let mut minutes: i64 = 0;
        let mut is_relative = false;

        for part in expr.split_whitespace() {
            if let Some(m) = part.strip_suffix('m') {
                if let Ok(val) = m.parse::<i64>() {
                    minutes += val;
                    is_relative = true;
                }
            } else if let Some(h) = part.strip_suffix('h') {
                if let Ok(val) = h.parse::<i64>() {
                    minutes += val * 60;
                    is_relative = true;
                }
            }
        }

        if is_relative && minutes > 0 {
            let target = now + Duration::minutes(minutes);
            return (
                target.timestamp(),
                target.format("%a, %b %d %I:%M %p").to_string(),
            );
        }

        let is_tomorrow = expr.contains("tomorrow");
        let target_date = if is_tomorrow {
            now.date_naive() + Duration::days(1)
        } else {
            now.date_naive()
        };

        if let Some(time_part) = expr.split_whitespace().find(|s| s.contains(':') || s.ends_with("am") || s.ends_with("pm")) {
            let mut hour = 9;
            let mut min = 0;

            if let Some((h_str, m_str)) = time_part.split_once(':') {
                let is_pm = m_str.ends_with("pm");
                let is_am = m_str.ends_with("am");
                let clean_m = m_str.trim_end_matches("am").trim_end_matches("pm");
                hour = h_str.parse().unwrap_or(9);
                min = clean_m.parse().unwrap_or(0);
                if is_pm && hour < 12 { hour += 12; }
                if is_am && hour == 12 { hour = 0; }
            } else if time_part.ends_with("pm") {
                hour = time_part.trim_end_matches("pm").parse().unwrap_or(9);
                if hour < 12 { hour += 12; }
            } else if time_part.ends_with("am") {
                hour = time_part.trim_end_matches("am").parse().unwrap_or(9);
                if hour == 12 { hour = 0; }
            }

            if let Some(target) = target_date.and_hms_opt(hour, min, 0) {
                let local_target = target.and_local_timezone(Local).unwrap();
                return (
                    local_target.timestamp(),
                    local_target.format("%a, %b %d %I:%M %p").to_string(),
                );
            }
        }

        let fallback = now + Duration::minutes(30);
        (fallback.timestamp(), fallback.format("%a, %b %d %I:%M %p").to_string())
    }

    pub fn update(&mut self, message: ReminderMessage) {
        match message {
            ReminderMessage::TaskInputChanged(txt) => {
                self.task_text = txt;
            }
            ReminderMessage::WhenInputChanged(w) => {
                self.when_text = w;
            }
            ReminderMessage::QuickWhen(w) => {
                self.when_text = w;
            }
            ReminderMessage::AddReminder => {
                let t = self.task_text.trim();
                let w = self.when_text.trim();

                    let (epoch, str_repr) = Self::parse_when(w);

                    });
                    self.task_text.clear();
                    self.when_text.clear();
                    self.save();
                }
            }
            ReminderMessage::DeleteReminder(id) => {
                self.entries.retain(|e| e.id != id);
                self.save();
            }
            ReminderMessage::Dismiss => {}
        }
    }

                self.cache.insert(word.to_lowercase(), defs);

    pub async fn lookup_word(word: &str) -> Vec<MeaningEntry> {
        let clean = word.trim().to_lowercase();
        if clean.is_empty() {
            return Vec::new();

        let Ok(out) = Command::new("curl")
            .args(["-s", "--max-time", "5", &url])
            .output()
            .await
        else {
            return Vec::new();

                                        part_of_speech: pos.clone(),
                                        definition: def_text,
                                    });
                                    if results.len() >= 12 {
                                        return results;
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

    }

impl WebSearchOverlay {

        Self { query: String::new() }
    }

    pub fn update(&mut self, message: WebSearchMessage) {
        if let WebSearchMessage::InputChanged(q) = message {
            self.query = q;
        }
    }

    pub fn build_url(raw: &str) -> Option<String> {
        let q = raw.trim();
        if q.is_empty() {
            return None;
        }

        let (base, clean_q) = if let Some(stripped) = q.strip_prefix("!g ") {
            ("https://www.google.com/search?q=", stripped)
        } else if let Some(stripped) = q.strip_prefix("!d ").or_else(|| q.strip_prefix("!ddg ")) {
            ("https://duckduckgo.com/?q=", stripped)
        } else if let Some(stripped) = q.strip_prefix("!w ").or_else(|| q.strip_prefix("!wiki ")) {
            ("https://en.wikipedia.org/wiki/Special:Search?search=", stripped)
        } else if let Some(stripped) = q.strip_prefix("!gh ") {
            ("https://github.com/search?q=", stripped)
        } else if let Some(stripped) = q.strip_prefix("!nix ") {
            ("https://search.nixos.org/packages?channel=unstable&query=", stripped)
        } else {
            ("https://www.startpage.com/sp/search?query=", q)
        };

        let encoded: String = clean_q
            .chars()
            .map(|c| match c {
                ' ' => "+".into(),
                'a'..='z' | 'A'..='Z' | '0'..='9' | '-' | '_' | '.' | '~' => c.to_string(),
                _ => format!("%{:02X}", c as u32),
            })
            .collect();

        Some(format!("{}{}", base, encoded))
    }

        let _ = Command::new("xdg-open").arg(url).spawn();
    }

            column![
                text("🌐 WEB SEARCH")
                    .size(theme.font_size_overlay)

                    .align_x(Alignment::Center),
                input,

                    c.dead = !c.dead;
                }
            }
            AmogusMessage::ToggleSafe(idx) => {
                if let Some(c) = self.crewmates.get_mut(idx) {
                    c.safe = !c.safe;
                    if c.safe {
                        c.impostor = false;
                        c.vented = false;

                if let Some(c) = self.crewmates.get_mut(idx) {
                    c.vented = !c.vented;
                }
            }

                for c in &mut self.crewmates {

                    c.impostor = false;
                    c.vented = false;

    pub summary: String,
}

#[derive(Debug, Clone)]
pub enum RngMessage {
    DiceCountChanged(String),
    StrengthChanged(String),
    FlatModChanged(String),
    SelectSides(u32),
    SelectMode(String),
    ExecuteRoll,
    Dismiss,
}

pub struct RngOverlay {
    pub dice_count: u32,
    pub selected_sides: u32,
    pub strength: i32,
    pub flat_mod: i32,
    pub keep_mode: String,
    pub last_total: i32,
    pub last_rolls: Vec<(i32, bool)>,
    pub last_type_str: String,
    pub history: Vec<RollHistoryItem>,
}

impl RngOverlay {
    pub fn new() -> Self {
        Self {
            dice_count: 1,
            selected_sides: 20,
            strength: 0,
            flat_mod: 0,
            keep_mode: "all".into(),
            last_total: 0,
            last_rolls: Vec::new(),
            last_type_str: String::new(),
            history: Vec::new(),
        }
    }

    fn random_u32() -> u32 {
        let mut buf = [0u8; 4];
        if let Ok(mut f) = File::open("/dev/urandom") {
            if f.read_exact(&mut buf).is_ok() {
                return u32::from_ne_bytes(buf);
            }
        }
        std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)
            .unwrap()
            .subsec_nanos()
    }

    pub fn roll(&mut self) {
        let count = self.dice_count.clamp(1, 100);
        let sides = self.selected_sides.clamp(2, 1000);

        if sides == 2 {
            let mut heads = 0;
            let mut tails = 0;
            let mut rolls = Vec::new();

            for _ in 0..count {
                if Self::random_u32() % 2 == 0 {
                    heads += 1;
                    rolls.push((1, true));
                } else {
                    tails += 1;
                    rolls.push((0, true));
                }
            }

            self.last_total = heads;
            self.last_rolls = rolls;
            self.last_type_str = format!("{} Coin Flips", count);
            let summary = format!("{} 🪙 Heads / {} 🪙 Tails", heads, tails);

            self.history.insert(
                0,
                RollHistoryItem {
                    time_str: Local::now().format("%H:%M:%S").to_string(),
                    spec: self.last_type_str.clone(),
                    total: heads,
                    summary,
                },
            );
            return;
        }

        let mut rolls: Vec<(i32, bool)> = (0..count)
            .map(|_| {
                let r = ((Self::random_u32() % sides) + 1) as i32;
                let adjusted = (r + self.strength).clamp(1, sides as i32) + self.flat_mod;
                (adjusted, true)
            })
            .collect();

        if self.keep_mode == "kh" {
            rolls.sort_by(|a, b| b.0.cmp(&a.0));
            for i in 1..rolls.len() {
                rolls[i].1 = false;
            }
        } else if self.keep_mode == "kl" {
            rolls.sort_by(|a, b| a.0.cmp(&b.0));
            for i in 1..rolls.len() {
                rolls[i].1 = false;
            }
        }

        let total: i32 = rolls.iter().filter(|r| r.1).map(|r| r.0).sum();
        let summary: Vec<String> = rolls
            .iter()
            .map(|r| {
                if r.1 {
                    r.0.to_string()
                } else {
                    format!("({})", r.0)
                }
            })
            .collect();

        self.last_total = total;
        self.last_rolls = rolls;
        self.last_type_str = format!("{}d{}", count, sides);

        self.history.insert(
            0,
            RollHistoryItem {
                time_str: Local::now().format("%H:%M:%S").to_string(),
                spec: self.last_type_str.clone(),
                total,
                summary: summary.join(", "),
            },
        );
    }

    pub fn update(&mut self, message: RngMessage) {
        match message {
            RngMessage::DiceCountChanged(val) => {
                self.dice_count = val.parse().unwrap_or(1).clamp(1, 100);
            }
            RngMessage::StrengthChanged(val) => {
                self.strength = val.parse().unwrap_or(0);
            }
            RngMessage::FlatModChanged(val) => {
                self.flat_mod = val.parse().unwrap_or(0);
            }
            RngMessage::SelectSides(s) => {
                self.selected_sides = s;
            }
            RngMessage::SelectMode(m) => {
                self.keep_mode = m;
            }
            RngMessage::ExecuteRoll => {
                self.roll();
            }
            RngMessage::Dismiss => {}
        }
    }

                .style(|_t, _s| button::Style { background: None, ..Default::default() });

        }

        let mode_row = row![
            text("Mode:").size(theme.font_size_normal).bold().color(theme.text),
            button(text("Keep All").size(12).color(if self.keep_mode == "all" { theme.success } else { theme.text }))
                .on_press(RngMessage::SelectMode("all".into()))
                .style(|_t, _s| button::Style { background: None, ..Default::default() }),
            button(text("Advantage").size(12).color(if self.keep_mode == "kh" { theme.success } else { theme.text }))

        )
        .on_press(RngMessage::ExecuteRoll)

use chrono::Local;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct NotificationToast {

    pub app_name: String,
    pub summary: String,
    pub body: String,
    pub timestamp: String,
    pub is_read: bool,
    pub is_urgent: bool,

    #[serde(skip, default = "Instant::now")]
    pub created_at: Instant,
}

#[derive(Debug, Clone)]
pub enum NotifMessage {
    DismissToast(u32),
    ClearHistory,
    DeleteHistory(u32),
    ToggleHistoryDrawer,
    DismissDrawer,
}

pub struct NotificationOverlay {
    pub active_toasts: Vec<NotificationToast>,
    pub history: Vec<NotificationToast>,
    pub show_history: bool,
}

impl NotificationOverlay {
    pub fn new() -> Self {
        Self {
            active_toasts: Vec::new(),
            history: Self::load_history(),
            show_history: false,
        }
    }

    fn history_path() -> PathBuf {
        let base = dirs::cache_dir().unwrap_or_else(|| PathBuf::from("/tmp"));
        base.join("moon_notification_history.json")
    }

    pub fn load_history() -> Vec<NotificationToast> {
        let path = Self::history_path();
        if path.exists() {
            if let Ok(raw) = fs::read_to_string(&path) {
                if let Ok(items) = serde_json::from_str::<Vec<NotificationToast>>(&raw) {
                    return items;
                }
            }
        }
        Vec::new()
    }

    pub fn save_history(&self) {
        let path = Self::history_path();
        if let Ok(data) = serde_json::to_string_pretty(&self.history) {
            let tmp = path.with_extension("tmp");
            if fs::write(&tmp, data).is_ok() {
                let _ = fs::rename(tmp, path);
            }
        }
    }

    pub fn push_notification(
        &mut self,
        app_name: &str,
        summary: &str,
        body: &str,
        is_urgent: bool,
        is_test: bool,
        settings: &ShellSettings,
    ) {
        let id = rand_id();

        // Match friend profile for custom border colors
        let matched_profile = settings.friend_profiles.iter().find(|p| {
            summary.to_lowercase().contains(&p.summary.to_lowercase())
                || app_name.to_lowercase().contains(&p.name.to_lowercase())
        });

        let border_color = if is_urgent {
            "#ff0000".to_string()
        } else if let Some(p) = matched_profile {
            p.color.clone()
        } else {
            settings.palette.base05.clone()
        };

        let item = NotificationToast {
            id,
            app_name: app_name.to_string(),
            summary: summary.to_string(),
            body: body.to_string(),
            timestamp: Local::now().format("%I:%M %p").to_string(),
            is_read: false,
            is_urgent,
            border_color,
            created_at: Instant::now(),
        };

        if self.active_toasts.len() >= 4 {
            self.active_toasts.remove(0);
        }
        self.active_toasts.push(item.clone());

        self.history.insert(0, item);
        if self.history.len() > 80 {
            self.history.truncate(80);
        }
        self.save_history();

        // Test notifications are completely silent; real notifications speak TTS if profile.tts is true
        if !is_test {
            if let Some(prof) = matched_profile {
                if prof.tts {
                    Self::speak_tts(summary, body, prof.default_readout.as_deref());
                }
            }
        }
    }

    fn speak_tts(summary: &str, body: &str, custom_readout: Option<&str>) {
        let s = summary.to_string();
        let b = body.to_string();
        let readout_template = custom_readout.unwrap_or("Message from {name}").to_string();

        std::thread::spawn(move || {
            let spoken = readout_template
                .replace("{name}", &s)
                .replace("{summary}", &s)
                .replace("{body}", &b);

            let _ = std::process::Command::new("sh")
                .args([
                    "-c",
                    "if command -v sage-tts >/dev/null 2>&1; then exec sage-tts \"$1\"; elif command -v spd-say >/dev/null 2>&1; then exec spd-say \"$1\"; elif command -v espeak-ng >/dev/null 2>&1; then exec espeak-ng \"$1\"; elif command -v espeak >/dev/null 2>&1; then exec espeak \"$1\"; fi",
                    "--",
                    &spoken,
                ])
                .spawn();
        });
    }

    pub fn tick_toasts(&mut self, hold_duration_sec: u64) {
        let hold_limit = std::time::Duration::from_secs(hold_duration_sec.max(1));
        self.active_toasts.retain(|t| t.is_urgent || t.created_at.elapsed() < hold_limit);
    }

    pub fn update(&mut self, message: NotifMessage) {
        match message {
            NotifMessage::DismissToast(id) => {
                self.active_toasts.retain(|t| t.id != id);
            }
            NotifMessage::ClearHistory => {
                self.history.clear();
                self.save_history();
            }
            NotifMessage::DeleteHistory(id) => {
                self.history.retain(|t| t.id != id);
                self.save_history();
            }
            NotifMessage::ToggleHistoryDrawer => {
                self.show_history = !self.show_history;
            }
            NotifMessage::DismissDrawer => {
                self.show_history = false;
            }
        }
    }

    pub fn view_toasts<'a>(&'a self, theme: &'a ShellTheme, settings: &'a ShellSettings) -> Element<'a, NotifMessage> {
        let card_w = settings.notif_card_width as f32;
        let card_h = settings.notif_card_height as f32;

        let shape = match settings.notif_shape.to_lowercase().as_str() {
            "hex" | "hexagon" => ShapeType::Hexagon { cut: 14 },
            "slant" => ShapeType::Slant { left: SlantSide::Right, right: SlantSide::Right },
            "flat" => ShapeType::Flat,
            _ => ShapeType::Rounded(10),
        };

        let mut deck = Stack::new();

        for (i, toast) in self.active_toasts.iter().enumerate() {
            let id = toast.id;
            let border_c = hex_to_color(&toast.border_color);

            // 60FPS Cubic Ease-Out Slide-In Animation Curve
            let elapsed_ms = toast.created_at.elapsed().as_millis() as f32;
            let slide_ms = (settings.notif_slide_in_ms.max(50)) as f32;
            let progress = (elapsed_ms / slide_ms).min(1.0);
            let ease_out = 1.0 - (1.0 - progress).powi(3);
            let slide_offset = (1.0 - ease_out) * (card_w + 60.0);

            // Stacking offset formula
            let y_step = card_h + (settings.notif_stack_overlap as f32);
            let y_pos = (i as f32) * y_step.max(10.0);

            let card_bg = ShapeBox::new()
                .shape(shape)
                .background(theme.background)
                .border(border_c, theme.border_width)
                .size(Length::Fixed(card_w), Length::Fixed(card_h));

            let badge_bg = if toast.is_urgent { theme.danger } else { border_c };
            let icon_badge = container(
                text(if toast.is_urgent { "!" } else { "i" })
                    .size(13)
                    .bold()
                    .color(Color::BLACK)
            )
            .padding([3, 7])
            .style(move |_t| container::Style {
                background: Some(iced::Background::Color(badge_bg)),
                border: iced::Border { radius: 10.0.into(), ..Default::default() },
                ..Default::default()
            });

    let mut b = [0u8; 4];
    if let Ok(mut f) = fs::File::open("/dev/urandom") {

        u32::from_ne_bytes(b)
    } else {
        1000
    }

pub struct LockscreenOverlay {

impl LockscreenOverlay {
    pub fn new() -> Self {
        Self {
            password_buffer: String::new(),
            is_locked: false,
            auth_failed: false,
            is_caps_lock: Self::check_caps_lock(),
        }

    pub fn check_caps_lock() -> bool {
        for i in 0..6 {
            let p = format!("/sys/class/leds/input{}::capslock/brightness", i);
            if let Ok(content) = std::fs::read_to_string(p) {
                if content.trim() == "1" {
                    return true;
                }
            }
        }
        false
    }

    pub async fn authenticate_pam(password: &str) -> bool {
        let user = std::env::var("USER").unwrap_or_else(|_| "root".into());
        let clean_pass = password.trim();
        if clean_pass.is_empty() {
            return false;
        }

        let child = Command::new("su")
            .args(["-c", "true", &user])
            .stdin(std::process::Stdio::piped())
            .stdout(std::process::Stdio::null())
            .stderr(std::process::Stdio::null())
            .spawn()
            .ok();

        if let Some(mut c) = child {
            use tokio::io::AsyncWriteExt;
            if let Some(mut stdin) = c.stdin.take() {
                let _ = stdin.write_all(format!("{}\n", clean_pass).as_bytes()).await;
            }
            if let Ok(st) = c.wait().await {
                return st.success();
            }
        }
        false
    }

    pub async fn set_dpms(on: bool) {
        let state = if on { "on" } else { "off" };
        if std::env::var("HYPRLAND_INSTANCE_SIGNATURE").is_ok() {
            let _ = Command::new("hyprctl").args(["dispatch", "dpms", state]).status().await;
        } else if std::env::var("SWAYSOCK").is_ok() {
            let _ = Command::new("swaymsg").args(["output", "*", "dpms", state]).status().await;
        }
    }

    pub fn update(&mut self, message: LockMessage) {
        match message {
            LockMessage::PasswordInputChanged(p) => {
                self.password_buffer = p;
                self.auth_failed = false;
                self.is_caps_lock = Self::check_caps_lock();
            }
            LockMessage::UnlockSuccess => {
                self.is_locked = false;
                self.password_buffer.clear();
                self.auth_failed = false;
                tokio::spawn(async {
                    Self::set_dpms(true).await;
                });
            }
            LockMessage::UnlockFailed => {
                self.password_buffer.clear();
                self.auth_failed = true;
            }
            _ => {}
        }
    }

    pub fn view<'a>(&'a self, theme: &ShellTheme) -> Element<'a, LockMessage> {
        let time_str = Local::now().format("%H:%M").to_string();
        let date_str = Local::now().format("%A, %B %d").to_string();

        } else {

        };

    }

    pub fn scan_desktop_files() -> Vec<AppEntry> {
        let mut apps = Vec::new();
        let home = dirs::home_dir().unwrap_or_else(|| PathBuf::from("/tmp"));
        let dirs = [
            PathBuf::from("/usr/share/applications"),
            PathBuf::from("/usr/local/share/applications"),
            home.join(".local/share/applications"),
            PathBuf::from("/run/current-system/sw/share/applications"),
            home.join(".nix-profile/share/applications"),
        ];

        let mut seen = std::collections::HashSet::new();

        for dir in &dirs {
            if let Ok(entries) = fs::read_dir(dir) {
                for entry in entries.flatten() {
                    let path = entry.path();
                    if path.extension().and_then(|s| s.to_str()) == Some("desktop") {
                        if let Ok(content) = fs::read_to_string(&path) {
                            let mut name = None;
                            let mut exec = None;
                            let icon = "🚀".to_string();
                            let mut nodisplay = false;

                            for line in content.lines() {
                                if line.starts_with("Name=") && name.is_none() {
                                    name = Some(line["Name=".len()..].trim().to_string());
                                } else if line.starts_with("Exec=") && exec.is_none() {
                                    let raw_exec = line["Exec=".len()..].trim();
                                    let cleaned = raw_exec
                                        .split('%')
                                        .next()
                                        .unwrap_or(raw_exec)
                                        .trim()
                                        .to_string();
                                    exec = Some(cleaned);
                                } else if line.eq_ignore_ascii_case("NoDisplay=true") {
                                    nodisplay = true;
                                }
                            }

                            if !nodisplay {
                                if let (Some(n), Some(e)) = (name, exec) {
                                    let key = format!("{}|{}", n.to_lowercase(), e.to_lowercase());
                                    if !seen.contains(&key) {
                                        seen.insert(key);
                                        apps.push(AppEntry { name: n, exec: e, icon });
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        apps.sort_by(|a, b| a.name.to_lowercase().cmp(&b.name.to_lowercase()));
        apps
    }

    pub fn update(&mut self, message: LauncherMessage) {
        if let LauncherMessage::SearchChanged(q) = message {
            self.search_query = q.clone();
            let lower = q.trim().to_lowercase();
            if lower.is_empty() {
                self.filtered_apps = self.all_apps.clone();
            } else {
                self.filtered_apps = self
                    .all_apps
                    .iter()
                    .filter(|a| {
                        a.name.to_lowercase().contains(&lower)
                            || a.exec.to_lowercase().contains(&lower)
                    })
                    .cloned()
                    .collect();
            }
        }
    }

    pub async fn launch(exec: &str) {
        if exec.is_empty() {
            return;
        }
        let _ = Command::new("systemd-run")
            .args(["--user", "--collect", "--quiet", "--slice=app.slice", "sh", "-c", exec])
            .spawn();
    }

                } else {

                },
            )
            .padding(14)
            .size(theme.font_size_overlay);

        let mut list_col = column![].spacing(6);

            SettingsMessage::NotifDropPause(v) => settings.notif_drop_pause_ms = v.clamp(0, 500),
            SettingsMessage::NotifDropDistance(v) => settings.notif_drop_distance_px = v.clamp(10, 500),
            SettingsMessage::NotifSummarySize(v) => settings.notif_summary_size = v.clamp(10, 36),
            SettingsMessage::NotifBodySize(v) => settings.notif_body_size = v.clamp(8, 32),
            SettingsMessage::NotifExcludedChanged(v) => settings.notif_excluded = v,

            SettingsMessage::SelectProfile(idx) => self.selected_profile = idx,
            SettingsMessage::ToggleProfileSound(idx) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.sound = !p.sound;
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::ToggleProfileTts(idx) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.tts = !p.tts;
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::ProfileSummaryChanged(idx, val) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.summary = val.clone();
                    p.name = val.to_lowercase().replace(' ', "_");
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::ProfileColorChanged(idx, col) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.color = col;
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::ProfileSoundPathChanged(idx, path) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.sound_path = if path.trim().is_empty() { None } else { Some(path.trim().to_string()) };
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::ProfileTtsTemplateChanged(idx, tpl) => {
                if let Some(p) = settings.friend_profiles.get_mut(idx) {
                    p.default_readout = if tpl.trim().is_empty() { None } else { Some(tpl.trim().to_string()) };
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::DeleteProfile(idx) => {
                if idx < settings.friend_profiles.len() {
                    settings.friend_profiles.remove(idx);
                    if self.selected_profile >= settings.friend_profiles.len() {
                        self.selected_profile = settings.friend_profiles.len().saturating_sub(1);
                    }
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::AddNewProfile => {
                settings.friend_profiles.push(FriendProfile {
                    name: "new_ping".into(),
                    summary: "New Ping".into(),
                    color: "#38FEDC".into(),
                    sound: true,
                    sound_path: None,
                    tts: true,
                    default_readout: Some("Message from {name}".into()),
                    content_rules: Vec::new(),
                });
                self.selected_profile = settings.friend_profiles.len().saturating_sub(1);
                let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
            }

            SettingsMessage::AddContentRule(p_idx) => {
                if let Some(p) = settings.friend_profiles.get_mut(p_idx) {
                    p.content_rules.push(ContentRule {
                        pattern: "logged in".into(),
                        readout: "{name} has logged in".into(),
                        sound_path: None,
                    });
                    let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                }
            }
            SettingsMessage::DeleteContentRule(p_idx, r_idx) => {
                if let Some(p) = settings.friend_profiles.get_mut(p_idx) {
                    if r_idx < p.content_rules.len() {
                        p.content_rules.remove(r_idx);
                        let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                    }
                }
            }
            SettingsMessage::RulePatternChanged(p_idx, r_idx, val) => {
                if let Some(p) = settings.friend_profiles.get_mut(p_idx) {
                    if let Some(r) = p.content_rules.get_mut(r_idx) {
                        r.pattern = val;
                        let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                    }
                }
            }
            SettingsMessage::RuleReadoutChanged(p_idx, r_idx, val) => {
                if let Some(p) = settings.friend_profiles.get_mut(p_idx) {
                    if let Some(r) = p.content_rules.get_mut(r_idx) {
                        r.readout = val;
                        let _ = crate::config::settings::save_friend_profiles(&settings.friend_profiles);
                    }
                }
            }

            SettingsMessage::ColorChanged(slot, hex) => {
                match slot.as_str() {
                    "base00" => settings.palette.base00 = hex,
                    "base01" => settings.palette.base01 = hex,
                    "base02" => settings.palette.base02 = hex,
                    "base03" => settings.palette.base03 = hex,
                    "base05" => settings.palette.base05 = hex,
                    "base08" => settings.palette.base08 = hex,
                    "base09" => settings.palette.base09 = hex,
                    "base0c" => settings.palette.base0c = hex,
                    "base0d" => settings.palette.base0d = hex,
                    _ => {}
                }
            }

            SettingsMessage::GpuTempWarn(v) => settings.gpu0.temp_warn = v,
            SettingsMessage::GpuTempDanger(v) => settings.gpu0.temp_danger = v,
            SettingsMessage::GpuVramWarn(v) => settings.gpu0.vram_warn = v,
            SettingsMessage::GpuVramDanger(v) => settings.gpu0.vram_danger = v,

            SettingsMessage::ToggleCapsuleVisible(id) => {
                let cur = settings.visible_capsules.get(&id).copied().unwrap_or(true);
                settings.visible_capsules.insert(id, !cur);
            }

            SettingsMessage::WindowWidth(id, w) => {
                let entry = settings.standalone_windows.entry(id).or_insert(crate::config::settings::WindowDim {
                    width: w,
                    height: 650,
                    field_height: None,
                    dismiss_on_blur: Some(true),
                    escape_closes: Some(true),
                });
                entry.width = w;
            }
            SettingsMessage::WindowHeight(id, h) => {
                let entry = settings.standalone_windows.entry(id).or_insert(crate::config::settings::WindowDim {
                    width: 840,
                    height: h,
                    field_height: None,
                    dismiss_on_blur: Some(true),
                    escape_closes: Some(true),
                });
                entry.height = h;
            }

            SettingsMessage::TestNotification => {
                settings.friend_profiles = crate::config::settings::load_friend_profiles();
            }
            SettingsMessage::SaveSettings => {}
            SettingsMessage::Dismiss => {}
        }

        let _ = settings.save();
    }

fn detect_standalone_overlay(args: &[String]) -> Option<OverlayTarget> {
    for (i, arg) in args.iter().enumerate() {
        if arg == "--overlay" || arg == "--ipc" {
            if let Some(target_str) = args.get(i + 1) {
                return OverlayTarget::from_str(target_str);
            }
        }
        if let Some(flag) = arg.strip_prefix("--") {
            if let Some(target) = OverlayTarget::from_str(flag) {
                return Some(target);
            }
        }
    }
    None
}

pub fn main() -> Result<(), iced_layershell::Error> {
    let args: Vec<String> = std::env::args().collect();

    // CLI Notification Push: moon-beauty-shell --notify "LusterDawn" "Message"
    if args.len() >= 3 && (args[1] == "--notify" || (args.len() >= 4 && args[1] == "--ipc" && args[2] == "notify")) {
        let rt = tokio::runtime::Runtime::new().unwrap();
        let summary = if args[1] == "--notify" { &args[2] } else { &args[3] };
        let body = if args[1] == "--notify" { args.get(3).map(|s| s.as_str()).unwrap_or("") } else { args.get(4).map(|s| s.as_str()).unwrap_or("") };
        let cmd = IpcCommand::IncomingNotif {
            app_name: "Desktop".into(),
            summary: summary.to_string(),
            body: body.to_string(),
            is_urgent: false,
        };
        let _ = rt.block_on(send_client_command(cmd));
        std::process::exit(0);
    }

    let standalone_target = detect_standalone_overlay(&args);
    let settings = ShellSettings::load();

    // Spawn the background D-Bus daemon if running main shell
    if standalone_target.is_none() {
        std::thread::spawn(|| {
            let daemon_script = "src/backend/notif_daemon.py";
            if std::path::Path::new(daemon_script).exists() {
                let _ = std::process::Command::new("python3").arg(daemon_script).spawn();
            }
        });
    }

    let binded_output_name = if settings.notif_screen_name != "Auto" && settings.notif_screen_name != "Auto (Primary)" && !settings.notif_screen_name.is_empty() {
        Some(settings.notif_screen_name.clone())
    } else {
        None
    };

    let layer_settings = if standalone_target.is_some() {
        LayerShellSettings {
            size: None,
            exclusive_zone: -1,
            anchor: Anchor::Top | Anchor::Bottom | Anchor::Left | Anchor::Right,
            layer: Layer::Overlay,
            keyboard_interactivity: KeyboardInteractivity::OnDemand,
            binded_output_name,
            ..Default::default()
        }
    } else {
        LayerShellSettings {
            size: Some((0, settings.bar_height)),
            exclusive_zone: settings.bar_height as i32,
            anchor: Anchor::Top | Anchor::Left | Anchor::Right,
            layer: Layer::Top,
            keyboard_interactivity: KeyboardInteractivity::None,
            binded_output_name,
            ..Default::default()
        }
    };

    MoonBeautyShellApp::run(Settings {
        layer_settings,
        ..Default::default()
    })
}

#[derive(Debug, Clone)]
pub enum Message {
    TickHardware,
    TickClock,
    TickToasts,
    MetricsCollected(CpuMetrics, RamMetrics, GpuMetrics, AudioStatus, Option<MprisTrack>),
    IpcReceived(IpcCommand),
    TopBar(TopBarMessage),
    Launcher(LauncherMessage),
    Calc(CalcMessage),
    Clipboard(ClipboardMessage),
    Notes(NotesMessage),
    Pass(PassMessage),
    Power(PowerMessage),
    Todo(TodoMessage),
    Reminders(ReminderMessage),
    Dict(DictMessage),
    Unicode(UnicodeMessage),
    Web(WebSearchMessage),
    Amogus(AmogusMessage),
    Rng(RngMessage),
    Quickshot(QuickshotMessage),
    Email(EmailMessage),
    Notif(NotifMessage),
    Settings(SettingsMessage),
    Lock(LockMessage),
    DismissActiveOverlay,
    CloseShell,
}

impl TryInto<iced_layershell::actions::LayershellCustomActionsWithInfo<()>> for Message {
    type Error = Message;
    fn try_into(self) -> Result<iced_layershell::actions::LayershellCustomActionsWithInfo<()>, Self::Error> {
        Err(self)
    }
}

        "moon-beauty-shell".into()
    }

    fn new(_flags: Self::Flags) -> (Self, Command<Message>) {

                weather: None,
                podman: Default::default(),
                borg: Default::default(),
                bar_time: ClockBackend::get_bar_time(),
            },
            active_overlay: standalone_target,
            is_standalone,

            launcher: AppLauncher::new(),
            calc: CalculatorOverlay::new(),
            clipboard: ClipboardOverlay::new(),
            notes: NotesOverlay::new(),
            pass: PassOverlay::new(),
            power: PowerOverlay::new(),
            todo: TodoOverlay::new(),

            dict: DictionaryOverlay::new(),

    }

    fn style(&self, _theme: &Self::Theme) -> iced_layershell::Appearance {
        iced_layershell::Appearance {
            background_color: Color::TRANSPARENT,
            text_color: self.theme.text,
        }
    }

    fn update(&mut self, message: Message) -> Command<Message> {
        match message {
            Message::TickClock => {
                self.telemetry.bar_time = ClockBackend::get_bar_time();
            }
            Message::TickToasts => {
                self.notifs.tick_toasts(self.settings.notif_hold_duration_sec);
            }
            Message::TickHardware => {
                let bat = SystemCollector::collect_battery();
                self.telemetry.battery = bat;

                return Command::perform(
                    async move {
                        let cpu = SystemCollector::collect_cpu().await;
                        let ram = SystemCollector::collect_ram().await;
                        let gpu = SystemCollector::collect_gpu("card0", false, 0).await;
                        let audio = AudioBackend::get_sink().await;
                        let mpris = MprisBackend::poll().await;
                        (cpu, ram, gpu, audio, mpris)
                    },
                    |(cpu, ram, gpu, audio, mpris)| {
                        Message::MetricsCollected(cpu, ram, gpu, audio, mpris)
                    },
                );
            }
            Message::MetricsCollected(cpu, ram, gpu, audio, mpris) => {
                self.telemetry.cpu = cpu;
                self.telemetry.ram = ram;
                self.telemetry.gpu = gpu;
                self.telemetry.audio = audio;
                self.telemetry.mpris = mpris;
            }
            Message::IpcReceived(cmd) => match cmd {
                IpcCommand::Toggle(target) => {
                    if self.active_overlay == Some(target.clone()) {
                        self.active_overlay = None;
                    } else {
                        self.active_overlay = Some(target);
                    }
                }
                IpcCommand::Open(target, _arg) => {
                    self.active_overlay = Some(target);
                }
                IpcCommand::Close(target) => {
                    if self.active_overlay == Some(target) {
                        self.active_overlay = None;
                    }
                }
                IpcCommand::CloseAll => {
                    self.active_overlay = None;
                }
                IpcCommand::DismissLatestNotif => {
                    if let Some(first) = self.notifs.active_toasts.first().cloned() {
                        self.notifs.update(NotifMessage::DismissToast(first.id));
                    }
                }
                IpcCommand::JumpToLatestNotif => {
                    self.active_overlay = Some(OverlayTarget::NotificationsHistory);
                }
                IpcCommand::ToggleHistory => {
                    if self.active_overlay == Some(OverlayTarget::NotificationsHistory) {
                        self.active_overlay = None;
                    } else {
                        self.active_overlay = Some(OverlayTarget::NotificationsHistory);
                    }
                }
                IpcCommand::ReloadSettings => {
                    self.settings = ShellSettings::load();
                    self.theme = ShellTheme::from_settings(&self.settings);
                }
                IpcCommand::TestNotification => {
                    // Send 3 notifications (first, second, third) completely silent
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #1 (first)", "Verification toast successfully routed.", false, true, &self.settings);
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #2 (second)", "Background worker service nominal.", false, true, &self.settings);
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #3 (third)", "Telemetry stream established.", false, true, &self.settings);
                }
                IpcCommand::IncomingNotif { app_name, summary, body, is_urgent } => {
                    self.notifs.push_notification(&app_name, &summary, &body, is_urgent, false, &self.settings);
                }
                IpcCommand::Ping => {}
            },
            Message::TopBar(msg) => {
                if let TopBarMessage::Action(act) = msg {
                    match act {
                        CapsuleAction::OpenOverlay(id) => {
                            let exe = std::env::current_exe().unwrap_or_else(|_| std::path::PathBuf::from("moon-beauty-shell"));
                            let _ = std::process::Command::new(exe).arg(format!("--{}", id)).spawn();
                        }
                        CapsuleAction::TogglePlayPause => {
                            return Command::perform(MprisBackend::toggle_play_pause(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        CapsuleAction::NextTrack => {
                            return Command::perform(MprisBackend::next(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        CapsuleAction::PreviousTrack => {
                            return Command::perform(MprisBackend::previous(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        CapsuleAction::CycleAudioSink => {
                            return Command::perform(AudioBackend::cycle_sinks(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        CapsuleAction::ToggleSinkMute => {
                            return Command::perform(AudioBackend::toggle_sink_mute(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        CapsuleAction::ToggleSourceMute => {
                            return Command::perform(AudioBackend::toggle_source_mute(), |_| Message::TopBar(TopBarMessage::Noop));
                        }
                        _ => {}
                    }
                }
            }
            Message::Launcher(msg) => match msg {
                LauncherMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                LauncherMessage::ExecuteApp(cmd) => {
                    if let Some(target_str) = cmd.strip_prefix("internal:") {
                        if let Some(target) = OverlayTarget::from_str(target_str) {
                            self.active_overlay = Some(target);
                            return Command::none();
                        }
                    }
                    if self.is_standalone {
                        tokio::spawn(async move { AppLauncher::launch(&cmd).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { AppLauncher::launch(&cmd).await; }, |_| Message::DismissActiveOverlay);
                }
                _ => self.launcher.update(msg),
            },
            Message::Calc(msg) => match msg {
                CalcMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                CalcMessage::CopyResult(val) => {
                    if self.is_standalone {
                        tokio::spawn(async move { CalculatorOverlay::copy_to_clipboard(&val).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { CalculatorOverlay::copy_to_clipboard(&val).await; }, |_| Message::DismissActiveOverlay);
                }
                _ => self.calc.update(msg),
            },
            Message::Clipboard(msg) => match msg {
                ClipboardMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                ClipboardMessage::CopyEntry(id) => {
                    if self.is_standalone {
                        tokio::spawn(async move { ClipboardOverlay::copy(id).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { ClipboardOverlay::copy(id).await; }, |_| Message::DismissActiveOverlay);
                }
                ClipboardMessage::DeleteEntry(id) => {
                    return Command::perform(async move { ClipboardOverlay::delete(id).await; }, |_| Message::DismissActiveOverlay);
                }
                ClipboardMessage::WipeAll => {
                    return Command::perform(async move { ClipboardOverlay::wipe().await; }, |_| Message::DismissActiveOverlay);
                }
                _ => {}
            },
            Message::Notes(msg) => match msg {
                NotesMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.notes.update(msg),
            },
            Message::Pass(msg) => match msg {
                PassMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                PassMessage::CopyKey(k) => {
                    if self.is_standalone {
                        tokio::spawn(async move { PassOverlay::copy_password(&k).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { PassOverlay::copy_password(&k).await; }, |_| Message::DismissActiveOverlay);
                }
                _ => self.pass.update(msg),
            },
            Message::Power(msg) => match msg {
                PowerMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => {
                    self.power.update(msg);
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
            },
            Message::Todo(msg) => match msg {
                TodoMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.todo.update(msg),
            },
            Message::Reminders(msg) => match msg {
                ReminderMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.reminders.update(msg),
            },
            Message::Dict(msg) => match msg {
                DictMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                DictMessage::SearchWord => {
                    let w = self.dict.word_query.clone();
                    return Command::perform(async move { (w.clone(), DictionaryOverlay::lookup_word(&w).await) }, |(w, defs)| {
                        Message::Dict(DictMessage::DefinitionsLoaded(w, defs))
                    });
                }
                _ => self.dict.update(msg),
            },
            Message::Unicode(msg) => match msg {
                UnicodeMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                UnicodeMessage::CopySymbol(s) => {
                    if self.is_standalone {
                        tokio::spawn(async move { UnicodeOverlay::copy(s).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { UnicodeOverlay::copy(s).await; }, |_| Message::DismissActiveOverlay);
                }
                _ => self.unicode.update(msg),
            },
            Message::Web(msg) => match msg {
                WebSearchMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                WebSearchMessage::ExecuteSearch => {
                    if let Some(url) = WebSearchOverlay::build_url(&self.web.query) {
                        if self.is_standalone {
                            tokio::spawn(async move { WebSearchOverlay::launch_browser(&url).await; });
                            std::process::exit(0);
                        }
                        self.active_overlay = None;
                        return Command::perform(async move { WebSearchOverlay::launch_browser(&url).await; }, |_| Message::DismissActiveOverlay);
                    }
                }
                _ => self.web.update(msg),
            },
            Message::Amogus(msg) => match msg {
                AmogusMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                AmogusMessage::TriggerScan => {
                    self.amogus.update(AmogusMessage::TriggerScan);
                    return Command::perform(AmogusOverlay::run_scan(), |cols| Message::Amogus(AmogusMessage::ScanFinished(cols)));
                }
                _ => self.amogus.update(msg),
            },
            Message::Rng(msg) => match msg {
                RngMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.rng.update(msg),
            },
            Message::Quickshot(msg) => match msg {
                QuickshotMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                QuickshotMessage::CaptureScreen => {
                    let r = self.quickshot.watermark_recipient.clone();
                    if self.is_standalone {
                        tokio::spawn(async move { QuickshotOverlay::capture_and_save(&r).await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(async move { QuickshotOverlay::capture_and_save(&r).await; }, |_| Message::DismissActiveOverlay);
                }
                QuickshotMessage::TriggerOcr => {
                    if self.is_standalone {
                        tokio::spawn(async move { QuickshotOverlay::run_ocr_screen().await; });
                        std::process::exit(0);
                    }
                    self.active_overlay = None;
                    return Command::perform(QuickshotOverlay::run_ocr_screen(), |_| Message::DismissActiveOverlay);
                }
                _ => self.quickshot.update(msg),
            },
            Message::Email(msg) => match msg {
                EmailMessage::Dismiss => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.email.update(msg),
            },
            Message::Notif(msg) => match msg {
                NotifMessage::DismissDrawer => {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.notifs.update(msg),
            },
            Message::Settings(msg) => {
                let is_test = matches!(msg, SettingsMessage::TestNotification);

                if is_test {
                    // Send 3 cascade items locally (silent) and broadcast over IPC
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #1 (first)", "Verification toast successfully routed.", false, true, &self.settings);
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #2 (second)", "Background worker service nominal.", false, true, &self.settings);
                    self.notifs.push_notification("Diagnostics", "System Diagnostics #3 (third)", "Telemetry stream established.", false, true, &self.settings);

                    if self.is_standalone {
                        tokio::spawn(async {
                            let _ = send_client_command(IpcCommand::TestNotification).await;
                        });
                    }
                } else if matches!(msg, SettingsMessage::Dismiss) {
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                } else {
                    self.settings_ui.update(msg, &mut self.settings);
                    self.theme = ShellTheme::from_settings(&self.settings);

                    if self.is_standalone {
                        tokio::spawn(async {
                            let _ = send_client_command(IpcCommand::ReloadSettings).await;
                        });
                    }
                }
            }
            Message::Lock(msg) => match msg {
                LockMessage::SubmitUnlock => {
                    let pass = self.lockscreen.password_buffer.clone();
                    return Command::perform(async move { LockscreenOverlay::authenticate_pam(&pass).await }, |ok| {
                        if ok { Message::Lock(LockMessage::UnlockSuccess) } else { Message::Lock(LockMessage::UnlockFailed) }
                    });
                }
                LockMessage::UnlockSuccess => {
                    self.lockscreen.update(msg);
                    if self.is_standalone { std::process::exit(0); }
                    self.active_overlay = None;
                }
                _ => self.lockscreen.update(msg),
            },
            Message::DismissActiveOverlay => {
                if self.is_standalone { std::process::exit(0); }
                self.active_overlay = None;
            }
            Message::CloseShell => {
                std::process::exit(0);
            }
        }
        Command::none()
    }

    fn subscription(&self) -> iced::Subscription<Message> {
        let clock_sub = iced::time::every(std::time::Duration::from_millis(1000)).map(|_| Message::TickClock);
        let toast_sub = iced::time::every(std::time::Duration::from_millis(16)).map(|_| Message::TickToasts);
        let hw_sub = iced::time::every(std::time::Duration::from_millis(self.settings.hardware_poll_interval_ms)).map(|_| Message::TickHardware);

        let esc_sub = iced::keyboard::on_key_press(|key, _modifiers| {
            if key == iced::keyboard::Key::Named(iced::keyboard::key::Named::Escape) {
                Some(Message::DismissActiveOverlay)
            } else {
                None
            }
        });

        let ipc_sub = iced::Subscription::run(|| {
            iced::stream::channel(20, |mut output| async move {
                if let Ok(server) = IpcServer::bind() {
                    loop {
                        let accept_result = server.accept().await;
                        match accept_result {
                            Ok((_, cmd)) => {
                                use iced::futures::SinkExt;
                                let _ = output.send(Message::IpcReceived(cmd)).await;
                            }
                            Err(_) => break,
                        }
                    }
                }
            })
        });

        iced::Subscription::batch(vec![clock_sub, toast_sub, hw_sub, esc_sub, ipc_sub])
    }

    fn view(&self) -> Element<'_, Message> {
        let is_right_corner = self.settings.notif_corner.contains("right");
        let is_top_corner = self.settings.notif_corner.contains("top");

        let pad_top = if is_top_corner { self.settings.notif_baseline_y as f32 } else { 0.0 };
        let pad_bottom = if !is_top_corner { self.settings.notif_baseline_y as f32 } else { 0.0 };
        let pad_right = if is_right_corner { self.settings.notif_margin_x.max(16) as f32 } else { 0.0 };
        let pad_left = if !is_right_corner { self.settings.notif_margin_x.max(16) as f32 } else { 0.0 };

        let notifs_view = container(self.notifs.view_toasts(&self.theme, &self.settings).map(Message::Notif))
            .width(Length::Fill)
            .height(Length::Fill)
            .align_x(if is_right_corner { iced::alignment::Horizontal::Right } else { iced::alignment::Horizontal::Left })
            .align_y(if is_top_corner { iced::alignment::Vertical::Top } else { iced::alignment::Vertical::Bottom })
            .padding(Padding {
                top: pad_top,
                right: pad_right,
                bottom: pad_bottom,
                left: pad_left,
            });

        let overlay_element: Option<Element<'_, Message>> = match &self.active_overlay {
            Some(OverlayTarget::Launcher) => Some(self.launcher.view(&self.theme, &self.settings).map(Message::Launcher)),
            Some(OverlayTarget::Calc) => Some(self.calc.view(&self.theme, &self.settings).map(Message::Calc)),
            Some(OverlayTarget::Clipboard) => Some(self.clipboard.view(&self.theme, &self.settings).map(Message::Clipboard)),
            Some(OverlayTarget::Notes) => Some(self.notes.view(&self.theme).map(Message::Notes)),
            Some(OverlayTarget::Pass) => Some(self.pass.view(&self.theme).map(Message::Pass)),
            Some(OverlayTarget::Power) => Some(self.power.view(&self.theme).map(Message::Power)),
            Some(OverlayTarget::Todo) => Some(self.todo.view(&self.theme).map(Message::Todo)),
            Some(OverlayTarget::Reminders) => Some(self.reminders.view(&self.theme).map(Message::Reminders)),
            Some(OverlayTarget::Dictionary) => Some(self.dict.view(&self.theme).map(Message::Dict)),
            Some(OverlayTarget::Unicode) => Some(self.unicode.view(&self.theme).map(Message::Unicode)),
            Some(OverlayTarget::Web) => Some(self.web.view(&self.theme).map(Message::Web)),
            Some(OverlayTarget::Amogus) => Some(self.amogus.view(&self.theme).map(Message::Amogus)),
            Some(OverlayTarget::Rng) => Some(self.rng.view(&self.theme).map(Message::Rng)),
            Some(OverlayTarget::Quickshot) => Some(self.quickshot.view(&self.theme).map(Message::Quickshot)),
            Some(OverlayTarget::Email) => Some(self.email.view(&self.theme).map(Message::Email)),
            Some(OverlayTarget::NotificationsHistory) => Some(self.notifs.view_history(&self.theme).map(Message::Notif)),
            Some(OverlayTarget::Settings) => Some(self.settings_ui.view(&self.settings, &self.theme).map(Message::Settings)),
            Some(OverlayTarget::Lockscreen) => Some(self.lockscreen.view(&self.theme).map(Message::Lock)),
            None => None,
        };

        if self.is_standalone {
            if let Some(overlay) = overlay_element {
                return Stack::new().push(overlay).push(notifs_view).into();
            } else {
                std::process::exit(0);
            }
        }

        let bar_view = TopBar::view(&self.settings, &self.theme, &self.telemetry).map(Message::TopBar);
        let base_content = Stack::new().push(bar_view).push(notifs_view);

        if let Some(overlay) = overlay_element {
            Stack::new().push(base_content).push(overlay).into()
        } else {
            base_content.into()
        }
    }

    fn theme(&self) -> Self::Theme {
        Theme::Dark
    }
}
